#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <EEPROM.h>

// ============== DISPLAY ===============
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 32
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

// =============== PIN ===============
const int TRIG = 5;
const int TOUCH1 = 3;
const int TOUCH2 = 4;
const int MOS1 = 1;
const int MOS2 = 2;

// =============== TEMPI SALDATURA ===============
// VARIABILI (Default caricate se EEPROM vuota)
unsigned int mainTime = 10, preTime = 3, preDelay = 10, boostDelay = 10, doubleDelay = 15, boostTime = 5, doubleTime = 10, weldDelay = 1000;

// =============== AUTO e LOCK ===============
const unsigned int autoDelay = 3000;
const unsigned int trigLock = 1200;

// =============== STATI ===============
enum Mode {OFF, NORMAL, NOPRE, BOOST, DOUBLEP};
Mode mode = OFF;

bool autoMode = false;
bool welding = false;
bool setupMenu = false;

unsigned long lastWeld = 0;

// =============== TIMER ===============
unsigned long trigStart = 0;
unsigned long t1Press = 0;
unsigned long t2Press = 0;

bool trigHeld=false;
bool autoToggleDone=false;

const int LONG_PRESS = 700;
const int MENU_HOLD = 5000;
const int AUTO_HOLD = 5000;

//===============  MENU ===============
int menuIndex = 0;
const int MENU_ITEMS = 7;
const char* menuNames[MENU_ITEMS] = {
"PRE TIME", "PRE DELAY", "BOOST DELAY", "DOUBLE DELAY", "BOOST TIME", "DOUBLE TIME", "WELD DELAY"
};

void setup() {

pinMode(MOS1,OUTPUT);
pinMode(MOS2,OUTPUT);

digitalWrite(MOS1,HIGH);
digitalWrite(MOS2,HIGH);

pinMode(TRIG,INPUT_PULLUP);
pinMode(TOUCH1,INPUT);
pinMode(TOUCH2,INPUT);

Wire.begin(8,9);
display.begin(SSD1306_SWITCHCAPVCC,0x3C);
display.setRotation(2);

EEPROM.begin(64);
loadSettings();

updateDisplay();
}

void loop() {

handleTrig();
handleTouch1();
handleTouch2();

if(autoMode && !welding && !setupMenu){
if(millis()-lastWeld > autoDelay){
weld();
}
}
}

// ================= TRIG =================
void handleTrig(){
bool state = digitalRead(TRIG)==LOW;
if(state && !trigHeld){
trigHeld=true;
trigStart=millis();
autoToggleDone=false;
}
if(state && trigHeld){
if(!autoToggleDone && millis()-trigStart>AUTO_HOLD){
autoMode=!autoMode;
autoToggleDone=true;

updateDisplay();
}
}
if(!state && trigHeld){
if(!setupMenu){
if(!autoToggleDone){
if(millis()-lastWeld>weldDelay){
weld();
}
}
}
trigHeld=false;
}
}

// ================= TOUCH1 =================
void handleTouch1(){
bool state = digitalRead(TOUCH1);
static bool held=false;
if(state && !held){
held=true;
t1Press=millis();
}
if(!state && held){
unsigned long pressTime = millis()-t1Press;
if(!setupMenu){
mode = (Mode)((mode+1)%5);
}else{
menuIndex++;
if(menuIndex>=MENU_ITEMS) menuIndex=0;
}
held=false;
updateDisplay();
}
}

// ================= TOUCH2 =================
void handleTouch2(){
bool state = digitalRead(TOUCH2);
static bool held=false;
if(state && !held){
held=true;
t2Press=millis();
}
if(state && held){
if(millis()-t2Press>MENU_HOLD){
setupMenu=!setupMenu;
if (!setupMenu) saveSettings();
autoMode=false;
welding=false;
held=false;
updateDisplay();
}
}
if(!state && held){
unsigned long pressTime = millis()-t2Press;
if(setupMenu){
changeMenuValue();
}else{
if(pressTime<LONG_PRESS){
mainTime++;
if(mainTime>20) mainTime=5;0;
}
}
held=false;
updateDisplay();
}
}

// ================= SOTTO MENU =================
void changeMenuValue(){
switch(menuIndex){
case 0:
preTime++;
if(preTime>5) preTime=1;
break;
case 1:
preDelay++;
if(preDelay>15) preDelay=5;
break;
case 2:
boostDelay+=1;
if(boostDelay>15) boostDelay=5;
break;
case 3:
doubleDelay++;
if(doubleDelay>20) doubleDelay=5;
break;
case 4:
boostTime++;
if(boostTime>5) boostTime=1;
break;
case 5:
doubleTime++;
if(doubleTime>15) doubleTime=5;
break;
case 6:
weldDelay+=200;
if(weldDelay>5000) weldDelay=400;
break;
}
}

// ================= SALDATURA =================
void weld(){
if(mode==OFF) return;
welding=true;
lastWeld=millis();
updateDisplay();
if(mode!=NOPRE){
pulse(preTime);
preciseDelay(preDelay);
}
pulse(mainTime);
if(mode==BOOST){
preciseDelay(boostDelay);
pulse(boostTime);
}
if(mode==DOUBLEP){
preciseDelay(doubleDelay);
pulse(doubleTime);
}
welding=false;
updateDisplay();
}

// ================= IMPULSO =================
void pulse(unsigned int ms){
if(ms > 40) ms = 40;
digitalWrite(MOS1,LOW);
digitalWrite(MOS2,LOW);
preciseDelay(ms);
digitalWrite(MOS1,HIGH);
digitalWrite(MOS2,HIGH);
delayMicroseconds(50);
}

// ================= DELAY PRECISO =================
void preciseDelay(unsigned int ms){
unsigned long start=micros();
while(micros()-start < ms*1000);{
if(digitalRead(TRIG) == HIGH){
digitalWrite(MOS1, HIGH);
digitalWrite(MOS2, HIGH);
welding = false;
return;
}
}
}

// ================= DISPLAY SOTTO MENU =================
void updateDisplay(){
display.clearDisplay();
display.setTextSize(1);
display.setTextColor(SSD1306_WHITE);
display.setCursor(0,0);

if(setupMenu){
display.println("SETUP MODE");
display.println(menuNames[menuIndex]);
display.setTextSize(2);
switch(menuIndex){
case 0: display.print(preTime); break;
case 1: display.print(preDelay); break;
case 2: display.print(boostDelay); break;
case 3: display.print(doubleDelay); break;
case 4: display.print(boostTime); break;
case 5: display.print(doubleTime); break;
case 6: display.print(weldDelay); break;
}
display.println(" ms");
}else{

// ================= DISPLAY =================

if (welding) display.println("*** WELDING ***");
else display.println("SPOT WELDER");
display.print("MODE ");
switch(mode){
case OFF: display.println("OFF"); break;
case NORMAL: display.println("NORMAL"); break;
case NOPRE: display.println("NO PRE"); break;
case BOOST: display.println("BOOST"); break;
case DOUBLEP: display.println("DOUBLE"); break;
}
  display.print("PRE:");
  display.print(preTime);
  display.println(" MAIN:");
  display.print("B:");
  display.print(boostTime);
  display.print(" D:");
  display.print(doubleTime);
  display.setCursor(75,16);
  display.setTextSize(2);
  display.print(mainTime);
  display.println("ms");
  if(autoMode){
    display.setTextSize(1);
    display.setCursor(90,0);
    display.print("AUTO");
}
}
display.display();
}

void saveSettings() {
  EEPROM.put(0, mainTime); EEPROM.put(4, preTime);
  EEPROM.put(8, preDelay); EEPROM.put(12, boostDelay);
  EEPROM.put(16, doubleDelay); EEPROM.put(20, boostTime);
  EEPROM.put(24, doubleTime); EEPROM.put(28, weldDelay);
  EEPROM.commit();
}

void loadSettings() {
  unsigned int check;
  EEPROM.get(0, check);
  if (check > 0 && check < 200) {
    EEPROM.get(4, preTime); EEPROM.get(8, preDelay);
    EEPROM.get(12, boostDelay); EEPROM.get(16, doubleDelay);
    EEPROM.get(20, boostTime); EEPROM.get(24, doubleTime);
    EEPROM.get(28, weldDelay);
  }
}
